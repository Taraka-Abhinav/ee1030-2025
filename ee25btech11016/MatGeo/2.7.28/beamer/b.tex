\begin{frame}{Question}
\[
\text{Find the area of the triangle whose vertices are }\\ \\
A(2,3),\; B(3,5),\; C(4,4).
\]
\end{frame}

\begin{frame}{The Vertices as Vectors}
\begin{table}[H]    
  \centering
  \begin{tabular}{|c|c|}
  \hline
  \textbf{Point} & \textbf{Vector} \\
  \hline
  A & $\myvec{2\\3\\0}$ \\
  \hline
  B & $\myvec{3\\5\\0}$ \\
  \hline
  C & $\myvec{4\\4\\0}$ \\
  \hline
  \end{tabular}
  \caption{Vectors}
  \label{Answers}
\end{table}
\end{frame}

\begin{frame}{The Sides of the Triangle}
\begin{align}
(\vec{A}-\vec{B})
&= \myvec{2\\3\\0}
  - \myvec{3\\5\\0}
= \myvec{-1\\-2\\0},
\label{eq:AB} \\[6pt]
(\vec{A}-\vec{C})
&= \myvec{2\\3\\0}
  - \myvec{4\\4\\0}
= \myvec{-2\\-1\\0}.
\label{eq:AC}
\end{align}
\end{frame}

\begin{frame}{Cross Product}
Using \eqref{eq:AB} and \eqref{eq:AC}, the magnitude of the cross product is
\begin{align}
\big\lVert (\vec{A}-\vec{B}) \times (\vec{A}-\vec{C}) \big\rVert
&= \sqrt{0^2 + 0^2 + (-3)^2} = 3.
\label{eq:norm}
\end{align}
\end{frame}

\begin{frame}{Area of Triangle}
Therefore, the area of triangle \(ABC\) is
\begin{align}
\text{ar}(\triangle ABC)
&= \tfrac{1}{2}\,\big\lVert (\vec{A}-\vec{B}) \times (\vec{A}-\vec{C}) \big\rVert
= \tfrac{1}{2}\times 3 = 1.5.
\label{eq:area}
\end{align}
\end{frame}

\begin{frame}[fragile]
    \frametitle{C Code}
    \begin{lstlisting}
#include <stdio.h>
#include <math.h>

// Function to compute cross product magnitude in 3D
double area_triangle(double A[3], double B[3], double C[3]) {
    double AB[3] = {A[0] - B[0], A[1] - B[1], 0};
    double AC[3] = {A[0] - C[0], A[1] - C[1], 0};
    double cross[3] = {
        AB[1]*AC[2] - AB[2]*AC[1],
        AB[2]*AC[0] - AB[0]*AC[2],
        AB[0]*AC[1] - AB[1]*AC[0]
    };
    double mag = sqrt(cross[0]*cross[0] + cross[1]*cross[1] + cross[2]*cross[2]);
    return 0.5 * mag;
}
    \end{lstlisting}
\end{frame}

import numpy as np
import matplotlib.pyplot as plt

# Define points
A = np.array([2,3,0])
B = np.array([3,5,0])
C = np.array([4,4,0])

# Vectors
AB = A - B
AC = A - C

# Cross product & area
cross = np.cross(AB,AC)
area = 0.5 * np.linalg.norm(cross)
print("Area of triangle =", area)
    \end{lstlisting}
