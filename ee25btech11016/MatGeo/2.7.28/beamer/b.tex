\begin{frame}{Question}
Find the area of the triangle whose vertices are $A(2,3)$, $B(3,5)$, and $C(4,4)$.
\end{frame}

\begin{frame}{Method: Area using Determinant}
The area of a triangle with vertices $\vec{A}$, $\vec{B}$, and $\vec{C}$ can be found using the formula:
\[ \text{Area} = \frac{1}{2} \left| \det(\vec{A}-\vec{B}, \vec{A}-\vec{C}) \right| \]
The given vertices are:
\[ \vec{A} = \myvec{2 \\ 3}, \quad \vec{B} = \myvec{3 \\ 5}, \quad \vec{C} = \myvec{4 \\ 4} \]
\end{frame}

\begin{frame}{Step 1: Find the Side Vectors}
First, we calculate two vectors that form the sides of the triangle.
\begin{align}
    \vec{A}-\vec{B} &= \myvec{2 \\ 3} - \myvec{3 \\ 5} = \myvec{-1 \\ -2} \label{eq:vec_ab} \\
    \vec{A}-\vec{C} &= \myvec{2 \\ 3} - \myvec{4 \\ 4} = \myvec{-2 \\ -1} \label{eq:vec_ac}
\end{align}
\end{frame}

\begin{frame}{Step 2: Calculate the Determinant}
Next, we find the determinant of the matrix formed by these two vectors.
\begin{align}
    \det(\vec{A}-\vec{B}, \vec{A}-\vec{C}) &= \begin{vmatrix} -1 & -2 \\ -2 & -1 \end{vmatrix} \nonumber \\
    &= (-1)(-1) - (-2)(-2) \nonumber \\
    &= 1 - 4 = -3 \label{eq:determinant}
\end{align}
\end{frame}

\begin{frame}{Step 3: Find the Area}
The area is half the absolute value of the determinant.
\begin{align}
    \text{Area} = \frac{1}{2} |-3| = 1.5
\end{align}
Thus, the area of the triangle is 1.5 square units.
\end{frame}

#include <stdio.h>
#include <math.h>

// Function to compute area via cross product
double area_triangle(double A[3], double B[3], double C[3]) {
    double AB[3] = {B[0] - A[0], B[1] - A[1], 0};
    double AC[3] = {C[0] - A[0], C[1] - A[1], 0};
    double cross_z = AB[0]*AC[1] - AB[1]*AC[0];
    return 0.5 * fabs(cross_z);
}

int main() {
    double A[3] = {2,3,0};
    double B[3] = {3,5,0};
    double C[3] = {4,4,0};

    double area = area_triangle(A,B,C);
    printf("Area of triangle = %.2f\n", area);
    return 0;
}

import numpy as np
import matplotlib.pyplot as plt

# Define points as 3D vectors
A = np.array([2,3,0])
B = np.array([3,5,0])
C = np.array([4,4,0])

# Form side vectors
vec_AB = B - A
vec_AC = C - A

# Cross product & area
cross_product = np.cross(vec_AB, vec_AC)
area = 0.5 * np.linalg.norm(cross_product)
print(f"Area of triangle = {area:.2f}")
    \end{lstlisting}
\end{frame}
