\textbf{Question}:\\
\text{Find the area of the triangle whose vertices are }
A(2,3),\; B(3,5),\; \text{and } C(4,4).

\solution \\
\begin{table}[H]
  \centering
  \begin{tabular}{|c|c|}
  \hline
  \textbf{Point} & \textbf{Vector} \\
  \hline
  A & $\myvec{2\\3\\0}$ \\
  \hline
  B & $\myvec{3\\5\\0}$ \\
  \hline
  C & $\myvec{4\\4\\0}$ \\
  \hline
  \end{tabular}
  \caption{Vectors}
  \label{Answers}
\end{table}

\begin{align}
(\vec{A}-\vec{B})
&= \myvec{2\\3\\0}
  - \myvec{3\\5\\0}
= \myvec{-1\\-2\\0},
\label{eq:AB} \\[6pt]
(\vec{A}-\vec{C})
&= \myvec{2\\3\\0}
  - \myvec{4\\4\\0}
= \myvec{-2\\-1\\0}.
\label{eq:AC}
\end{align}

Using \eqref{eq:AB} and \eqref{eq:AC}, the magnitude of the cross product is
\begin{align}
\big\lVert (\vec{A}-\vec{B}) \times (\vec{A}-\vec{C}) \big\rVert
&= \sqrt{0^2 + 0^2 + ((-1)(-1) - (-2)(-2))^2} \nonumber \\
&= \sqrt{0^2 + 0^2 + (-3)^2} = 3.
\label{eq:norm}
\end{align}

Therefore the area of triangle \(ABC\) is
\begin{align}
\text{ar}(\triangle ABC)
&= \tfrac{1}{2}\,\big\lVert (\vec{A}-\vec{B}) \times (\vec{A}-\vec{C}) \big\rVert
= \tfrac{1}{2}\times 3 = 1.5.
\label{eq:area}
\end{align}

      Therefore, \(\displaystyle \text{ar}(\triangle ABC)=1.5\).
