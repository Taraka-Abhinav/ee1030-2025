\begin{frame}{Question}
Find the area of the region bounded by the lines $y=4x+5$, $y=5-x$, and $4y=x+5$.
\end{frame}

\begin{frame}{Step 1: Finding the Vertices}

\textbf{Vertex A:} Intersection of $y=4x+5$ and $y=5-x$
\begin{align}
    4x+5 = 5-x \implies 5x = 0 \implies x=0, y=5 \nonumber
\end{align}
$\vec{A} = \myvec{0 \\ 5}$

\textbf{Vertex B:} Intersection of $y=5-x$ and $4y=x+5$
\begin{align}
    4(5-x) = x+5 \implies 20-4x = x+5 \implies x=3, y=2 \nonumber
\end{align}
$\vec{B} = \myvec{3 \\ 2}$

\textbf{Vertex C:} Intersection of $y=4x+5$ and $4y=x+5$
\begin{align}
    4(4x+5) = x+5 \implies 16x+20 = x+5 \implies x=-1, y=1 \nonumber
\end{align}
$\vec{C} = \myvec{-1 \\ 1}$
\end{frame}

\begin{frame}{Step 2: Calculating the Area}
We use the determinant formula for the area:
\[
\text{Area} = \frac{1}{2} \left| \det(\vec{A}-\vec{B}, \vec{A}-\vec{C}) \right|
\]
First, we find the side vectors:
\begin{align}
    \vec{A}-\vec{B} &= \myvec{0 \\ 5} - \myvec{3 \\ 2} = \myvec{-3 \\ 3} \label{eq:side1} \\
    \vec{A}-\vec{C} &= \myvec{0 \\ 5} - \myvec{-1 \\ 1} = \myvec{1 \\ 4} \label{eq:side2}
\end{align}
\end{frame}

\begin{frame}{Step 2: Calculating the Area (continued)}
Now, we calculate the determinant:
\begin{align}
    \det(\vec{A}-\vec{B}, \vec{A}-\vec{C}) &= \begin{vmatrix} -3 & 1 \\ 3 & 4 \end{vmatrix} \nonumber \\
    &= (-3)(4) - (1)(3) \nonumber \\
    &= -12 - 3 = -15 \label{eq:det_val}
\end{align}
The area is half the absolute value of the determinant:
\begin{align}
    \text{Area} = \frac{1}{2} |-15| = 7.5
\end{align}
The area of the triangle is $7.5$ square units.
\end{frame}

#include <stdio.h>
#include <stdlib.h>
#include <math.h> // For fabs

double triangle_area() {
    double Ax = 0, Ay = 5;
    double Bx = 3, By = 2;
    double Cx = -1, Cy = 1;
    // Area formula using determinant concept
    double area = 0.5 * ((Ax - Cx)*(By - Ay) - 
                         (Ax - Bx)*(Cy - Ay));
    return fabs(area);
}

int main() {
    double area = triangle_area();
    printf("Area of triangle = %.4f\n", area);
    return 0;
}

\end{frame}

import numpy as np
import matplotlib.pyplot as plt

# Triangle vertices
A = np.array([0, 5])
B = np.array([3, 2])
C = np.array([-1, 1])
triangle = np.array([A, B, C, A])

# Area calculation
area = 0.5 * abs(np.linalg.det(np.column_stack((A-B, A-C))))
print(f"Area of triangle = {area:.2f}")

# Plot triangle
plt.plot(triangle[:,0], triangle[:,1], 'b-', linewidth=2)
plt.fill(triangle[:,0], triangle[:,1], color='cyan', alpha=0.3)
    \end{lstlisting}
\end{frame}
