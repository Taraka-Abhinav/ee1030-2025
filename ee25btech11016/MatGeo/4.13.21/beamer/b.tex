\begin{frame}{Question}
The X coordinate of the incentre of the triangle that has the coordinates of mid points of its sides as (0, 1), (1, 1) and (1, 0) is: \hfill (2013)

\begin{enumerate}
    \item $2 + \sqrt{2}$
    \item $2 - \sqrt{2}$
    \item $1 + \sqrt{2}$
    \item $1 - \sqrt{2}$
\end{enumerate}
\end{frame}

\begin{frame}{Step 1: Finding the Vertices from Midpoints}
Let the position vectors of the midpoints of sides $BC$, $AC$, and $AB$ be $\vec{D}$, $\vec{E}$, and $\vec{F}$ respectively.
\begin{align}
\vec{D} = \myvec{0 \\ 1}, \quad \vec{E} = \myvec{1 \\ 1}, \quad \vec{F} = \myvec{1 \\ 0}
\end{align}

The position vectors of the vertices of the triangle, $\vec{A}$, $\vec{B}$, and $\vec{C}$, can be found from the midpoint vectors.
\begin{align}
\vec{A} &= \vec{E} + \vec{F} - \vec{D} = \myvec{1 \\ 1} + \myvec{1 \\ 0} - \myvec{0 \\ 1} = \myvec{2 \\ 0} \\
\vec{B} &= \vec{D} + \vec{F} - \vec{E} = \myvec{0 \\ 1} + \myvec{1 \\ 0} - \myvec{1 \\ 1} = \myvec{0 \\ 0} \\
\vec{C} &= \vec{D} + \vec{E} - \vec{F} = \myvec{0 \\ 1} + \myvec{1 \\ 1} - \myvec{1 \\ 0} = \myvec{0 \\ 2}
\end{align}

Next, we find the lengths of the sides of $\triangle ABC$. Let $a$, $b$, and $c$ be the lengths of the sides opposite to vertices $A$, $B$, and $C$.
\end{frame}

\begin{frame}[fragile]
    \frametitle{Solution}
\begin{align}
a &= \norm{\vec{C} - \vec{B}} = \norm{\myvec{0 \\ 2}} = \sqrt{0^2 + 2^2} = 2 \\
b &= \norm{\vec{A} - \vec{C}} = \norm{\myvec{2 \\ -2}} = \sqrt{2^2 + (-2)^2} = \sqrt{8} = 2\sqrt{2} \\
c &= \norm{\vec{B} - \vec{A}} = \norm{\myvec{-2 \\ 0}} = \sqrt{(-2)^2 + 0^2} = 2
\end{align}

The position vector of the incentre, $\vec{I}$, is given by the formula:
\begin{align}
\vec{I} = \frac{a\vec{A} + b\vec{B} + c\vec{C}}{a+b+c}
\end{align}
\end{frame}

\begin{frame}[fragile]
    \frametitle{Solution}
Substituting the vectors and side lengths:
\begin{align}
\vec{I} &= \frac{2\myvec{2 \\ 0} + 2\sqrt{2}\myvec{0 \\ 0} + 2\myvec{0 \\ 2}}{2 + 2\sqrt{2} + 2} \\
    &= \frac{1}{4 + 2\sqrt{2}} \left( \myvec{4 \\ 0} + \myvec{0 \\ 0} + \myvec{0 \\ 4} \right) \\
    &= \frac{1}{4 + 2\sqrt{2}} \myvec{4 \\ 4} = \myvec{\frac{4}{4 + 2\sqrt{2}} \\ \frac{4}{4 + 2\sqrt{2}}}
\end{align}
The x-coordinate of the incentre, $I_x$, is the first component of $\vec{I}$.
\begin{align}
I_x &= \frac{4}{4 + 2\sqrt{2}} = \frac{2}{2 + \sqrt{2}} \\
    &= \frac{2(2 - \sqrt{2})}{(2 + \sqrt{2})(2 - \sqrt{2})} = \frac{2(2 - \sqrt{2})}{4-2} = 2 - \sqrt{2}
\end{align}
\end{frame}

#include <stdio.h>
#include <math.h>

typedef struct {
    double x, y;
} Point;

Point make_point(double x, double y) {
    Point p; p.x = x; p.y = y; return p;
}

double dist(Point p1, Point p2) {
    return sqrt(pow(p1.x - p2.x,2) + pow(p1.y - p2.y,2));
}

Point find_incentre(Point A, Point B, Point C) {
    double a = dist(B,C);
    double b = dist(A,C);
    double c = dist(A,B);
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]
    \frametitle{C Code - Verification}
    \begin{lstlisting}
    Point I;
    I.x = (a*A.x + b*B.x + c*C.x) / (a+b+c);
    I.y = (a*A.y + b*B.y + c*C.y) / (a+b+c);
    return I;
}

int main() {
    // Vertices derived from midpoints
    Point A = make_point(2,0);
    Point B = make_point(0,0);
    Point C = make_point(0,2);

    Point I = find_incentre(A,B,C);
    printf("Numeric Incentre: (%.4f, %.4f)\n", I.x, I.y);
    printf("Value of 2-sqrt(2) is approx %.4f\n", 2-sqrt(2));
    return 0;
}

import numpy as np
import matplotlib.pyplot as plt

# Midpoints
D = np.array([0,1])
E = np.array([1,1])
F = np.array([1,0])

# Vertices using midpoint relations

B = D + F - E   # (0,0)

# Side lengths
a = np.linalg.norm(C-B)
b = np.linalg.norm(A-C)
c = np.linalg.norm(A-B)
