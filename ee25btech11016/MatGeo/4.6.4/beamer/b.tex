\begin{frame}{Question}
Find the equation of a line passing through the point $(2, 3, 2)$ and parallel to the line:
\[ \vec{r} = \myvec{-2 \\ 3 \\ 0} + \lambda\myvec{2 \\ -3 \\ 6} \]
Also, find the distance between these two lines.
\end{frame}

\begin{frame}{Step 1: Equation of the Required Line}
The vector equation of a line is $\vec{r} = \vec{a} + \mu\vec{b}$.

The line passes through the point $(2, 3, 2)$, so the position vector is:
\begin{align}
    \vec{a_2} = \myvec{2 \\ 3 \\ 2}
\end{align}
The line is parallel to the given line, so they share the same direction vector:
\begin{align}
    \vec{b} = \myvec{2 \\ -3 \\ 6}
\end{align}
The equation of the required line is:
\begin{align}
    \vec{r} = \myvec{2 \\ 3 \\ 2} + \mu\myvec{2 \\ -3 \\ 6}
\end{align}
\end{frame}

\begin{frame}{Step 2: Distance Between Lines}
The distance $d$ between two parallel lines is given by the formula:
\begin{align}
    d = \frac{\|\vec{b} \times (\vec{a_2} - \vec{a_1})\|}{\|\vec{b}\|}
\end{align}
From the line equations, we have:
\begin{align}
    \vec{a_1} = \myvec{-2 \\ 3 \\ 0}, \quad \vec{a_2} = \myvec{2 \\ 3 \\ 2}, \quad \vec{b} = \myvec{2 \\ -3 \\ 6}
\end{align}
First, we find the difference vector $(\vec{a_2} - \vec{a_1})$:
\begin{align}
    \vec{a_2} - \vec{a_1} = \myvec{4 \\ 0 \\ 2}
\end{align}
\end{frame}

\begin{frame}{Step 3: Calculating the Distance}
Next, we compute the cross product:
\begin{align}
    \vec{b} \times (\vec{a_2} - \vec{a_1}) = \myvec{2 \\ -3 \\ 6} \times \myvec{4 \\ 0 \\ 2} = \myvec{-6 \\ 20 \\ 12}
\end{align}
Now, we find the norms (magnitudes):
\begin{align}
    \|\vec{b} \times (\vec{a_2} - \vec{a_1})\| &= \sqrt{(-6)^2 + 20^2 + 12^2} = \sqrt{580} \\
    \|\vec{b}\| &= \sqrt{2^2 + (-3)^2 + 6^2} = \sqrt{49} = 7
\end{align}
Finally, the distance is:
\begin{align}
    d = \frac{\sqrt{580}}{7} = \frac{2\sqrt{145}}{7} \approx 3.44
\end{align}
\end{frame}

import ctypes
import numpy as np
import matplotlib.pyplot as plt

# Load compiled shared library
lib = ctypes.CDLL('./line_distance.so')
lib.main()

# Given data
a1 = np.array([-2, 3, 0])
a2 = np.array([2, 3, 2])
b = np.array([2, -3, 6])

# Compute distance
da = a2 - a1
v = np.cross(b, da)
d = np.linalg.norm(v) / np.linalg.norm(b)
print("Distance =", d)
