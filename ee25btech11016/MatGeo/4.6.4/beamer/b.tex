\begin{frame}{Question}
Find the equation of a line passing through the point $(2, 3, 2)$ and parallel to the line:
\[ \vec{r} = \myvec{-2 \\ 3 \\ 0} + \lambda\myvec{2 \\ -3 \\ 6} \]
Also, find the distance between these two lines.
\end{frame}

\begin{frame}{Step 1: Equation of the Required Line}
The vector equation of a line is $\vec{r} = \vec{a_2} + \mu\vec{b}$.

The line passes through the point $(2, 3, 2)$, so the position vector is:
\begin{align}
    \vec{a_2} = \myvec{2 \\ 3 \\ 2}
\end{align}
The line is parallel to the given line, so they share the same direction vector:
\begin{align}
    \vec{b} = \myvec{2 \\ -3 \\ 6}
\end{align}
The equation of the required line is:
\begin{align}
    \vec{r} = \myvec{2 \\ 3 \\ 2} + \mu\myvec{2 \\ -3 \\ 6}
\end{align}
\end{frame}

\begin{frame}{Step 2: Distance via Least Squares}
Let $\Delta\vec{a} = \vec{a_2} - \vec{a_1}$. We want to find the minimum of the squared distance function $f(\nu)$:
\begin{align*}
    f(\nu) &= \|\Delta\vec{a} - \nu\vec{b}\|^2 \\
           &= (\Delta\vec{a} - \nu\vec{b})^T(\Delta\vec{a} - \nu\vec{b}) \\
           &= \Delta\vec{a}^T\Delta\vec{a} - 2\nu(\Delta\vec{a}^T\vec{b}) + \nu^2(\vec{b}^T\vec{b})
\end{align*}
To find the minimum, we set the derivative with respect to $\nu$ to zero.
\end{frame}

\begin{frame}{Step 3: Minimizing the Distance}
Taking the derivative of $f(\nu)$ and setting it to zero:
\begin{align}
    \frac{df}{d\nu} = -2(\Delta\vec{a}^T\vec{b}) + 2\nu(\vec{b}^T\vec{b}) = 0
\end{align}
Solving for the optimal value of $\nu$:
\begin{align}
    \nu_{opt} = \frac{\Delta\vec{a}^T\vec{b}}{\vec{b}^T\vec{b}}
\end{align}
Substituting this back into the distance function gives the minimum distance squared.
\end{frame}

\begin{frame}{Step 4: Final Distance Formula}
The minimum squared distance is:
\begin{align}
    d^2 = \frac{(\Delta\vec{a}^T\Delta\vec{a})(\vec{b}^T\vec{b}) - (\Delta\vec{a}^T\vec{b})^2}{\vec{b}^T\vec{b}}
\end{align}
Using Lagrange's identity, the numerator simplifies to $\|\Delta\vec{a} \times \vec{b}\|^2$.
\begin{align}
    d^2 = \frac{\|\Delta\vec{a} \times \vec{b}\|^2}{\|\vec{b}\|^2} \implies d = \frac{\|\Delta\vec{a} \times \vec{b}\|}{\|\vec{b}\|}
\end{align}
\end{frame}

\begin{frame}{Step 5: Numerical Calculation}
We substitute the numerical values:
\begin{align}
    \Delta\vec{a} = \myvec{4 \\ 0 \\ 2} \quad , \quad \vec{b} = \myvec{2 \\ -3 \\ 6}
\end{align}
The cross product is:
\begin{align}
    \Delta\vec{a} \times \vec{b} = \myvec{6 \\ -20 \\ -12}
\end{align}
The norms are:
\begin{align}
    \|\Delta\vec{a} \times \vec{b}\| &= \sqrt{6^2 + (-20)^2 + (-12)^2} = \sqrt{580} \\
    \|\vec{b}\| &= \sqrt{2^2 + (-3)^2 + 6^2} = 7
\end{align}
The distance is:
\begin{align}
    d = \frac{\sqrt{580}}{7} = \frac{2\sqrt{145}}{7}
\end{align}
\end{frame}

    double bnorm = sqrt(b[0]*b[0] + b[1]*b[1] + b[2]*b[2]);

import ctypes
import numpy as np
import matplotlib.pyplot as plt

# Load compiled shared library
lib = ctypes.CDLL('./line_distance.so')
lib.main()

# Given data
a1 = np.array([-2, 3, 0])
a2 = np.array([2, 3, 2])
b = np.array([2, -3, 6])

# Compute distance
da = a2 - a1
v = np.cross(b, da)
d = np.linalg.norm(v) / np.linalg.norm(b)
print("Distance =", d)
