\textbf{Question}:\\
Find the equation of a line passing through the point $(2, 3, 2)$ and parallel to the line $\vec{r} = (-2\hat{i} + 3\hat{j}) + \lambda(2\hat{i} - 3\hat{j} + 6\hat{k})$. Also, find the distance between these two lines.

\solution \\

\textbf{1. Equation of the Line}\\
The equation of the line passing through point $\vec{a_2}$ parallel to vector $\vec{b}$ is:
\begin{align}
    \vec{r} = \vec{a_2} + \mu\vec{b} = \myvec{2 \\ 3 \\ 2} + \mu\myvec{2 \\ -3 \\ 6}
\end{align}

\textbf{2. Distance Between Lines using Least Squares}\\
Let $\Delta\vec{a} = \vec{a_2} - \vec{a_1}$. The squared distance between the lines is the minimum of the function:
\begin{align}
    f(\nu) = \|\Delta\vec{a} - \nu\vec{b}\|^2 = (\Delta\vec{a} - \nu\vec{b})^T(\Delta\vec{a} - \nu\vec{b}) \nonumber \\
    = \Delta\vec{a}^T\Delta\vec{a} - 2\nu(\Delta\vec{a}^T\vec{b}) + \nu^2(\vec{b}^T\vec{b})
\end{align}
To find the minimum, we set the derivative to zero:
\begin{align}
    \frac{df}{d\nu} = -2(\Delta\vec{a}^T\vec{b}) + 2\nu(\vec{b}^T\vec{b}) = 0 \implies \nu_{opt} = \frac{\Delta\vec{a}^T\vec{b}}{\vec{b}^T\vec{b}}
\end{align}
Substituting $\nu_{opt}$ back gives the minimum squared distance:
\begin{align}
    d^2 = \frac{(\Delta\vec{a}^T\Delta\vec{a})(\vec{b}^T\vec{b}) - (\Delta\vec{a}^T\vec{b})^2}{\vec{b}^T\vec{b}}
\end{align}
By Lagrange's identity, the numerator is $\|\Delta\vec{a} \times \vec{b}\|^2$. This gives the distance formula:
\begin{align}
    d = \frac{\|\Delta\vec{a} \times \vec{b}\|}{\|\vec{b}\|}
\end{align}
The numerical values are:
\begin{align}
    \vec{a_1} = \myvec{-2 \\ 3 \\ 0}, \quad \vec{a_2} = \myvec{2 \\ 3 \\ 2}, \quad \vec{b} = \myvec{2 \\ -3 \\ 6}
\end{align}
\begin{align}
    \Delta\vec{a} = \vec{a_2} - \vec{a_1} = \myvec{4 \\ 0 \\ 2}
\end{align}
The cross product is:
\begin{align}
    \Delta\vec{a} \times \vec{b} = \myvec{4 \\ 0 \\ 2} \times \myvec{2 \\ -3 \\ 6} = \myvec{6 \\ -20 \\ -12}
\end{align}
The norms are:
\begin{align}
    \|\Delta\vec{a} \times \vec{b}\| &= \sqrt{6^2 + (-20)^2 + (-12)^2} = \sqrt{580} \\
    \|\vec{b}\| &= \sqrt{2^2 + (-3)^2 + 6^2} = 7
\end{align}
The distance is:
\begin{align}
    d = \frac{\sqrt{580}}{7} = \frac{2\sqrt{145}}{7}
\end{align}
